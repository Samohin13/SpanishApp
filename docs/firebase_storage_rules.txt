rules_version = '2';
service firebase.storage {
  match /b/{bucket}/o {

    // Аватары пользователей: только владелец может писать
    match /users/{uid}/{file=**} {
      allow read:  if true;
      allow write: if request.auth != null && request.auth.uid == uid;
    }

    // Контент-пакеты лежат в корне бакета как *.json:
    // публичное чтение, запись запрещена из приложения.
    match /{filename} {
      allow read:  if filename.matches('.*\\.json');
      allow write: if false;
    }
  }
}
