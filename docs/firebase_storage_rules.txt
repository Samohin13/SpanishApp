rules_version = '2';
service firebase.storage {
  match /b/{bucket}/o {

    // Аватары пользователей: только владелец может писать
    match /users/{uid}/{file=**} {
      allow read:  if true;
      allow write: if request.auth != null && request.auth.uid == uid;
    }

    // Контент-пакеты: публичное чтение, запись запрещена из приложения
    // (заливаем пакеты только через Firebase Console)
    match /content_packs/{file=**} {
      allow read:  if true;
      allow write: if false;
    }

    // По умолчанию всё остальное закрыто
    match /{path=**} {
      allow read:  if false;
      allow write: if false;
    }
  }
}
